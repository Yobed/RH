4. Implementer la fonction `handleGenererAttestation`:
```typescript
async function handleGenererAttestation(type: "travail" | "salaire") {
  if (!company || !employee) return;
  setGeneratingType(type);
  try {
    const supabase = createBrowserClient();

    // Generer PDF
    const doc = generateAttestationPDF({
      employee: {
        full_name: employee.full_name,
        poste: employee.poste ?? "—",
        date_embauche: employee.date_embauche,
        salaire_brut: employee.salaire_brut,
        matricule: employee.matricule,
      },
      company,
      type,
    });

    // Telecharger
    exportPDF(doc, `attestation_${type}_${employee.full_name.replace(/\s+/g, "_")}`);

    // Archiver dans Storage
    const pdfBlob = doc.output("blob");
    const fileName = `attestation_${type}_${employee.id}_${Date.now()}.pdf`;
    const storagePath = `${company.id}/${employee.id}/${fileName}`;

    const { error: storageError } = await supabase.storage
      .from("documents")
      .upload(storagePath, pdfBlob, { contentType: "application/pdf" });

    if (!storageError) {
      const { data: { publicUrl } } = supabase.storage
        .from("documents")
        .getPublicUrl(storagePath);

      const label = type === "travail" ? "travail" : "salaire";
      const { data: newDoc } = await supabase
        .from("documents")
        .insert({
          company_id: company.id,
          employee_id: employee.id,
          name: `Attestation de ${label} — ${employee.full_name} — ${new Date().toLocaleDateString("fr-CI")}`,
          famille: "Attestation",
          file_type: "application/pdf",
          file_url: publicUrl,
        })
        .select("id, name, famille, file_type, file_size_kb, created_at, file_url")
        .single();

      if (newDoc) {
        setLocalDocuments((prev) => [newDoc, ...prev]);
      }
    }
  } finally {
    setGeneratingType(null);
  }
}
```
