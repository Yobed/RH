**Partie B — app/api/career-events/route.ts:**

Remplacer le contenu actuel par une version avec validation Zod:

```typescript
import { createServerClient } from "@/lib/supabase/server";
import { NextResponse } from "next/server";
import { z } from "zod";

export const dynamic = "force-dynamic";

const careerEventSchema = z.object({
  employee_id: z.string().uuid("employee_id doit etre un UUID valide"),
  company_id: z.string().uuid("company_id doit etre un UUID valide"),
  event_type: z.enum(["promotion", "mutation", "augmentation", "formation", "avenant", "autre"], {
    errorMap: () => ({ message: "Type d'evenement invalide" }),
  }),
  date_event: z.string().regex(/^\d{4}-\d{2}-\d{2}$/, "Format date invalide (YYYY-MM-DD)"),
  description: z.string().max(500).nullable().optional(),
  new_value: z.record(z.unknown()).nullable().optional(),
  old_value: z.record(z.unknown()).nullable().optional(),
});

export async function POST(req: Request) {
  try {
    const supabase = createServerClient();
    const body: unknown = await req.json();

    const parsed = careerEventSchema.safeParse(body);
    if (!parsed.success) {
      return NextResponse.json(
        { error: parsed.error.errors[0]?.message ?? "Donnees invalides" },
        { status: 400 }
      );
    }

    const { error } = await supabase.from("career_events").insert(parsed.data);

    if (error) throw error;

    return NextResponse.json({ success: true });
  } catch (error: unknown) {
    const message = error instanceof Error ? error.message : "Erreur serveur";
    return NextResponse.json({ error: message }, { status: 500 });
  }
}
```

Note: `z.record(z.unknown())` au lieu de `any` pour le JSONB — conforme a la regle no-any du projet.
