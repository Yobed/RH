```typescript
// lib/paie-sage-import.ts
// Fonctions utilitaires pour l'import du Livre de Paie Sage
// Adaptées de app/api/import/employees/route.ts (parseSalary)

/**
 * Nettoie et parse une valeur numérique au format français/Sage.
 * Gère : espaces insécables (\xa0,  ), virgules décimales, séparateurs de milliers (points).
 * Retourne null si la valeur est invalide ou absente (contrairement à parseSalary qui retourne 0).
 */
export function parseNumericFR(val: unknown): number | null {
  if (typeof val === "number") return isNaN(val) ? null : val;
  if (val === null || val === undefined || val === "") return null;
  // Supprime espaces insécables (\xa0), espaces fines insécables ( ), espaces ordinaires
  let s = String(val).replace(/[\s  ﻿]/g, "");
  // Convertit la virgule décimale en point
  s = s.replace(",", ".");
  // Gestion milliers avec points (ex: 1.500.000 → 1500000)
  const parts = s.split(".");
  if (parts.length > 2 || (parts.length === 2 && parts[1].length === 3)) {
    s = parts.join("");
  }
  const n = parseFloat(s);
  return isNaN(n) ? null : n;
}

export const REQUIRED_COLUMNS = ["Matricule/Nom", "Salaire de base", "Net A Payer"] as const;

/**
 * Vérifie que toutes les colonnes requises sont présentes dans la première ligne du fichier.
 * Retourne un message d'erreur pour la première colonne manquante, ou null si tout est présent.
 */
export function validateRequiredColumns(
  required: readonly string[],
  firstRow: Record<string, unknown>
): string | null {
  for (const col of required) {
    if (!(col in firstRow)) {
      return `Colonne '${col}' manquante dans le fichier importé. Vérifiez que vous avez utilisé le template fourni.`;
    }
  }
  return null;
}
```
